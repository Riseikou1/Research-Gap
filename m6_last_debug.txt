def _tokens(text: str) -> list[str]:
    normalized = canonical_evidence_key(text)

    normalized = re.sub(
        r"\b(?:other|unknown|misc|miscellaneous|unspecified|"
        r"uncategorized|none)"
        r"(?:\s+(?:method|methods|model|models|approach|approaches|"
        r"technique|techniques|population|populations|setting|settings|"
        r"dataset|datasets|data|baseline|baselines))?\b",
        " ",
        normalized,
    )

    return [
        token
        for token in re.findall(r"[a-z0-9]+", normalized)
        if token not in _STOPWORDS
        and token not in _GENERIC_BUCKETS
    ]


def _concept_tokens(text: str) -> set[str]:
    """Normalize a few harmless grammatical variants."""

    forms = {
        "reduce": "reduce",
        "reduces": "reduce",
        "reduced": "reduce",
        "reducing": "reduce",
        "reduction": "reduce",
        "reductions": "reduce",

        "decrease": "reduce",
        "decreases": "reduce",
        "decreased": "reduce",
        "decreasing": "reduce",

        "improve": "improve",
        "improves": "improve",
        "improved": "improve",
        "improving": "improve",
        "improvement": "improve",
        "improvements": "improve",

        "increase": "increase",
        "increases": "increase",
        "increased": "increase",
        "increasing": "increase",
    }

    result: set[str] = set()

    for token in _tokens(text):
        token = forms.get(token, token)

        if len(token) > 5 and token.endswith("ies"):
            token = token[:-3] + "y"
        elif len(token) > 6 and token.endswith("tion"):
            token = token[:-3] + "t"
        elif len(token) > 5 and token.endswith("ing"):
            token = token[:-3]
        elif len(token) > 5 and token.endswith("ed"):
            token = token[:-2]
        elif len(token) > 5 and token.endswith("s"):
            token = token[:-1]

        result.add(token)

    return result


def _query_overlap(left: str, right: str) -> float:
    """Compare query content without interpreting scientific vocabulary."""

    left_terms = set(re.findall(r"[a-z0-9]+", left.casefold()))
    right_terms = set(re.findall(r"[a-z0-9]+", right.casefold()))

    if not left_terms or not right_terms:
        return 0.0

    return len(left_terms & right_terms) / min(len(left_terms), len(right_terms))


def _phrase_matches(value: str, text: str) -> bool:
    """Require substantial concept containment rather than loose overlap."""

    required = _concept_tokens(value)
    observed = _concept_tokens(text)

    if not required or not observed:
        return False

    return required <= observed or observed <= required


def deduplicate_query_phrases(
    phrases: Sequence[str],
) -> list[str]:
    """Remove repeated complete phrases while preserving their order."""

    result: list[str] = []
    seen: set[str] = set()
    for phrase in phrases:
        key = canonical_evidence_key(phrase)
        if not key or key in seen:
            continue
        seen.add(key)
        result.append(phrase)
    return result


def _short_phrase(
    *parts: str | Sequence[str],
    max_terms: int = 24,
) -> str:
    """Bound a query while deduplicating complete phrases only."""

    phrases: list[str] = []

    for part in parts:
        values = [part] if isinstance(part, str) else part

        for value in values:
            if not value:
                continue

            phrases.append(value)

    phrases = deduplicate_query_phrases(phrases)

    terms: list[str] = []
    for phrase in phrases:
        terms.extend(_tokens(phrase))

    return " ".join(
        terms[:max_terms]
    )[:1000].strip()

def _normalize_requirement(
    value: str,
    dimension: str,
) -> str:
    return canonical_evidence_key(value)


def _requirement_key(
    value: str,
    dimension: str,
) -> str:
    normalized = _normalize_requirement(
        value,
        dimension,
    )

    return normalized or canonical_evidence_key(value)

def _idea_canonical_value(
    idea: ResearchIdea,
    value: str,
    dimension: str,
) -> str | None:
    """Look up a provider-derived generic identity for one surface value."""

    facet = _CANONICAL_FACET_NAMES.get(dimension, dimension)
    values = idea.canonical_facets.get(facet, {})
    value_key = canonical_evidence_key(value)

    for surface, canonical in values.items():
        if canonical_evidence_key(surface) == value_key:
            return canonical

    return None


def _idea_requirement_key(
    idea: ResearchIdea,
    value: str,
    dimension: str,
) -> str:
    canonical = _idea_canonical_value(idea, value, dimension)
    return canonical_evidence_key(canonical or value)


def _canonical_item_matches(
    idea: ResearchIdea,
    value: str,
    item: EvidenceItem,
    dimension: str,
) -> bool:
    """Match upstream canonical identities before lexical fallbacks."""

    requested = _idea_canonical_value(idea, value, dimension)
    observed = item.canonical_value
    return _canonical_concept_matches(requested, observed)


def _canonical_concept_matches(
    left: str | None,
    right: str | None,
) -> bool:
    """Match canonical concepts by equality or conservative containment."""

    if not left or not right:
        return False

    left_key = canonical_evidence_key(left)
    right_key = canonical_evidence_key(right)

    if left_key == right_key:
        return True

    return _phrase_matches(left, right)


def _explicit_synonym_item_matches(
    idea: ResearchIdea,
    value: str,
    item: EvidenceItem,
    dimension: str,
) -> bool:
    """Match an evidence surface value through declared decomposer synonyms."""

    observed = " ".join(
        part
        for part in (
            item.canonical_value or "",
            item.value,
            item.evidence_text,
        )
        if part
    )
    linked_values = _linked_requirement_values(idea, value, dimension)
    return any(
        _phrase_matches(linked, observed)
        for linked in linked_values
    )


def _synonym_requirement_keys(
    idea: ResearchIdea,
    value: str,
    dimension: str,
) -> set[str]:
    """Return all explicit equivalents associated with one requirement."""

    keys = {
        _requirement_key(
            value,
            dimension,
        )
    }

    value_key = canonical_evidence_key(
        value
    )

    for canonical, alternatives in idea.synonyms.items():
        linked = [
            canonical,
            *alternatives,
        ]

        linked_raw_keys = {
            canonical_evidence_key(item)
            for item in linked
        }

        linked_requirement_keys = {
            _requirement_key(
                item,
                dimension,
            )
            for item in linked
        }

        if (
            value_key in linked_raw_keys
            or keys & linked_requirement_keys
        ):
            keys.update(
                linked_requirement_keys
            )

    return {
        key
        for key in keys
        if key
    }


def _linked_requirement_values(
    idea: ResearchIdea,
    value: str,
    dimension: str,
) -> list[str]:
    """Return one requirement plus explicitly declared equivalents."""

    result = [value]

    raw_key = canonical_evidence_key(
        value
    )
    requirement_key = _requirement_key(
        value,
        dimension,
    )

    for canonical, alternatives in idea.synonyms.items():
        linked = [
            canonical,
            *alternatives,
        ]

        raw_keys = {
            canonical_evidence_key(item)
            for item in linked
        }

        requirement_keys = {
            _requirement_key(
                item,
                dimension,
            )
            for item in linked
        }

        if (
            raw_key in raw_keys
            or requirement_key in requirement_keys
        ):
            result.extend(
                linked
            )

    unique: list[str] = []
    seen: set[str] = set()

    for item in result:
        key = canonical_evidence_key(
            item
        )

        if not key or key in seen:
            continue

        seen.add(
            key
        )
        unique.append(
            item
        )

    return unique


def _idea_requirement_groups(
    idea: ResearchIdea,
    values: Sequence[str],
    dimension: str,
) -> list[tuple[str, ...]]:
    """Keep distinct requirements separate while grouping explicit equivalents."""

    groups: list[list[str]] = []

    for value in values:
        if (
            not value
            or not is_concrete_entity(value)
        ):
            continue

        alternatives = _linked_requirement_values(
            idea,
            value,
            dimension,
        )

        alternative_keys = {
            _idea_requirement_key(
                idea,
                item,
                dimension,
            )
            for item in alternatives
        }

        matching_group: list[str] | None = None

        for group in groups:
            group_keys = {
                _idea_requirement_key(
                    idea,
                    item,
                    dimension,
                )
                for item in group
            }

            if alternative_keys & group_keys:
                matching_group = group
                break

        if matching_group is None:
            groups.append(
                alternatives
            )
        else:
            matching_group.extend(
                alternatives
            )

    return [
        tuple(
            dict.fromkeys(group)
        )
        for group in groups
    ]


def _all_requirement_groups_match(
    groups: Sequence[Sequence[str]],
    matcher,
) -> bool:
    return bool(groups) and all(
        any(
            matcher(value)
            for value in group
        )
        for group in groups
    )

def _facet_query_variants(
    idea: ResearchIdea,
    value: str,
    dimension: str,
) -> tuple[str, ...]:
    values = [
        value,
    ]

    normalized = _normalize_requirement(
        value,
        dimension,
    )

    if normalized:
        values.append(
            normalized
        )

    value_key = canonical_evidence_key(
        value
    )

    for canonical, alternatives in idea.synonyms.items():
        linked = [
            canonical,
            *alternatives,
        ]

        if value_key in {
            canonical_evidence_key(item)
            for item in linked
        }:
            values.extend(
                linked
            )

    return tuple(
        dict.fromkeys(
            item
            for item in values
            if _tokens(item)
        )
    )


def _query_terms_for_requirements(
    idea: ResearchIdea,
    values: Sequence[str],
    dimension: str,
) -> list[str]:
    groups = _idea_requirement_groups(
        idea,
        values,
        dimension,
    )

    parts: list[str] = []

    for group in groups:
        variants = _facet_query_variants(
            idea,
            group[0],
            dimension,
        )

        parts.extend(
            variants[:2]
        )

    return list(
        dict.fromkeys(
            parts,
        )
    )


def build_idea_verification_queries(
    idea: ResearchIdea,
    *,
    max_queries: int = 3,
) -> list[VerificationQuery]:
    """Build bounded searches preserving every explicit research requirement."""

    if not 1 <= max_queries <= 3:
        raise ValueError(
            "max_queries must be between 1 and 3"
        )

    method = _query_terms_for_requirements(
        idea,
        _idea_methods(idea),
        "method",
    )

    problem = _query_terms_for_requirements(
        idea,
        _idea_problems(idea),
        "problem",
    )

    context = _query_terms_for_requirements(
        idea,
        _idea_context_values(idea),
        "context",
    )

    modality = _query_terms_for_requirements(
        idea,
        idea.data_or_modality,
        "data_or_modality",
    )

    domain = _query_terms_for_requirements(
        idea,
        idea.domain,
        "context",
    )

    constraints = _query_terms_for_requirements(
        idea,
        _idea_constraints(idea),
        "constraint",
    )

    comparison = _query_terms_for_requirements(
        idea,
        idea.comparison,
        "comparison",
    )

    outcomes = _query_terms_for_requirements(
        idea,
        idea.outcomes,
        "outcome",
    )

    raw_queries = [
        _short_phrase(
            method,
            comparison,
            problem,
            outcomes,
            context,
            modality,
            domain,
            constraints,
        ),
        _short_phrase(
            problem,
            method,
            context,
            modality,
            domain,
            comparison,
            constraints,
            outcomes,
        ),
        _short_phrase(
            method,
            problem,
            modality,
            constraints,
            comparison,
            outcomes,
            context,
            domain,
        ),
    ]

    result: list[VerificationQuery] = []
    seen: set[str] = set()

    for text in raw_queries:
        key = " ".join(
            _tokens(text)
        )

        if (
            not key
            or key in seen
            or len(key.split()) < 2
        ):
            continue

        seen.add(
            key
        )

        result.append(
            VerificationQuery(
                candidate_id="idea",
                query=text,
                pattern_type="direct_idea_assessment",
                strategy="verification_direct_idea",
                source="deterministic",
            )
        )

        if len(result) >= max_queries:
            break

    return result


def _candidate_support_variants(
    candidate: GapCandidate,
    dimension: str,
    value: str,
) -> list[str]:
    """Recover wording variants from the candidate's own provenance."""

    target = _normalize_requirement(
        value,
        dimension,
    )

    result: list[str] = []

    for item in candidate.supporting_evidence:
        item_dimension = {
            "method_or_intervention": "method",
            "method": "method",
            "comparison_or_baseline": "comparison",
            "comparison": "comparison",
            "constraint": "constraint",
            "constraints": "constraint",
            "dataset": "dataset",
            "datasets": "dataset",
            "population_or_setting": "population_or_setting",
            "data_or_modality": "data_or_modality",
            "research_objective": "problem",
        }.get(
            item.evidence_type,
            item.evidence_type,
        )

        if dimension in {
            "method",
            "method_family",
        }:
            compatible = item_dimension == "method"
        else:
            compatible = (
                item_dimension == dimension
            )

        if not compatible:
            continue

        normalized = _normalize_requirement(
            item.value,
            dimension,
        )

        if (
            target
            and normalized
            and normalized == target
        ):
            result.append(
                item.value
            )

    return result


def _candidate_facet_groups(
    candidate: GapCandidate,
) -> list[tuple[str, ...]]:
    groups: list[tuple[str, ...]] = []

    for basis in candidate.landscape_basis:
        if (
            not basis.value
            or not _tokens(basis.value)
        ):
            continue

        values = [
            basis.value,
        ]

        normalized = _normalize_requirement(
            basis.value,
            basis.dimension,
        )

        if normalized:
            values.append(
                normalized
            )

        values.extend(
            _candidate_support_variants(
                candidate,
                basis.dimension,
                basis.value,
            )
        )

        groups.append(
            tuple(
                dict.fromkeys(
                    value
                    for value in values
                    if _tokens(value)
                )
            )
        )

    if groups:
        return groups

    fallback = " ".join(
        _tokens(candidate.title)
    )

    return [
        (fallback,)
    ] if fallback else []


def _query_preserves_candidate_facets(
    query: str,
    candidate: GapCandidate,
) -> bool:
    query_terms = set(
        _tokens(query)
    )

    for alternatives in _candidate_facet_groups(
        candidate
    ):
        if not any(
            set(_tokens(value)) <= query_terms
            for value in alternatives
        ):
            return False

    return True


def build_verification_queries(
    idea: ResearchIdea,
    candidate: GapCandidate,
    *,
    max_queries: int = 3,
) -> list[VerificationQuery]:
    """Build deterministic candidate-specific counterexample searches."""

    if not 1 <= max_queries <= 3:
        raise ValueError(
            "max_queries must be between 1 and 3"
        )

    basis_terms = " ".join(
        item.value
        for item in candidate.landscape_basis[:4]
    )

    candidate_facets = " ".join(
        group[0]
        for group in _candidate_facet_groups(candidate)
        if group
    )

    methods = " ".join(
        _idea_methods(idea)[:2]
    )

    problems = " ".join(
        _idea_problems(idea)[:2]
    )

    context = " ".join(
        _idea_context_values(idea)[:2]
    )

    modality = " ".join(idea.data_or_modality[:2])
    domain = " ".join(idea.domain[:2])

    pattern_terms = _PATTERN_TERMS.get(
        candidate.pattern_type,
        "direct evaluation",
    )

    raw_queries = [
        _short_phrase(
            candidate_facets,
            methods,
            problems,
            context,
            modality,
            domain,
            pattern_terms,
        ),
        _short_phrase(
            candidate_facets,
            candidate.title,
            problems,
            modality,
            pattern_terms,
        ),
        _short_phrase(
            candidate_facets,
            basis_terms,
            methods,
            modality,
            pattern_terms,
        ),
    ]

    original_key = " ".join(
        _tokens(idea.original_text)
    )

    result: list[VerificationQuery] = []
    seen: set[str] = set()

    for text in raw_queries:
        key = " ".join(
            _tokens(text)
        )

        if (
            not key
            or key == original_key
            or key in seen
            or len(key.split()) < 2
            or not _query_preserves_candidate_facets(
                text,
                candidate,
            )
        ):
            continue

        seen.add(
            key
        )

        result.append(
            VerificationQuery(
                candidate_id=candidate.id,
                query=text,
                pattern_type=candidate.pattern_type,
                strategy="verification_counterexample",
                source="deterministic",
            )
        )

        if len(result) >= max_queries:
            break

    return result

class GapVerifier:
    """Verify ideas and candidate gaps using targeted retrieval."""

    def __init__(
        self,
        retriever: MultiQueryRetriever,
        extractor: PaperExtractor,
        *,
        lexical_scorer: LexicalScorer | None = None,
        max_queries: int = 3,
        per_query_limit: int = 10,
        max_verification_papers: int = 10,
        max_verified_candidates: int = 4,
    ) -> None:
        if (
            not 1 <= max_queries <= 3
            or not 1 <= per_query_limit <= 100
            or not 1 <= max_verification_papers <= 100
            or not 1 <= max_verified_candidates <= 12
        ):
            raise ValueError(
                "verification limits are outside their allowed bounds"
            )

        self.retriever = retriever
        self.extractor = extractor
        self.lexical_scorer = (
            lexical_scorer
            or LexicalScorer()
        )
        self.max_queries = max_queries
        self.per_query_limit = per_query_limit
        self.max_verification_papers = max_verification_papers
        self.max_verified_candidates = max_verified_candidates
        self.notices: list[str] = []
        self._metrics: dict[str, int] = {
            "verification_evidence_requested": 0,
            "verification_papers_already_cached": 0,
            "new_verification_extractions": 0,
            "verification_queries_executed": 0,
            "verification_papers_prescreened": 0,
            "verification_papers_prescreen_accepted": 0,
            "verification_papers_prescreen_uncertain": 0,
            "verification_papers_rejected_by_prescreen": 0,
            "verification_queries_planned": 0,
            "verification_queries_cache_hits": 0,
            "verification_queries_skipped_redundant": 0,
            "verification_candidates_early_stopped": 0,
            "verification_candidates_reused": 0,
            "candidate_verification_results_reused_exact": 0,
            "candidate_verification_results_reused_nonidentical": 0,
            "observed_candidates_rejected_preverification": 0,
            "candidate_rejected_unanchored": 0,
            "candidate_rejected_observed": 0,
            "candidate_rejected_dominated": 0,
            "candidate_rejected_invalid_role": 0,
            "candidate_hypotheses_after_pruning": 0,
            "candidate_hypotheses_verified": 0,
        }
        self._verification_paper_ids: set[str] = set()
        self._verification_query_cache: dict[
            tuple[str, str, str, int, int], MultiQueryResult
        ] = {}
        self._verification_outcome_cache: dict[tuple[tuple[str, str], ...], GapVerification] = {}
        self._verification_evidence_cache: dict[tuple[str, str], PaperEvidence] = {}
        self._timings: dict[str, float] = {
            "direct_verification_retrieval": 0.0,
            "direct_verification_evidence_extraction": 0.0,
            "candidate_verification_retrieval": 0.0,
            "candidate_verification_evidence_extraction": 0.0,
        }

    def metrics_snapshot(self) -> dict[str, int]:
        """Return cumulative verification work counters."""

        return {
            **self._metrics,
            "verification_unique_papers": len(self._verification_paper_ids),
        }

    def timings_snapshot(self) -> dict[str, float]:
        return dict(self._timings)

    def reset_metrics(self) -> None:
        """Start a fresh per-pipeline-run accounting window."""

        for key in self._metrics:
            self._metrics[key] = 0
        self._verification_paper_ids.clear()
        self._verification_query_cache.clear()
        self._verification_outcome_cache.clear()
        self._verification_evidence_cache.clear()
        for key in self._timings:
            self._timings[key] = 0.0

    def prime_evidence(
        self,
        papers: Sequence[Paper],
        evidence: Sequence[PaperEvidence],
    ) -> None:
        """Make exact initial-pass evidence available to verification."""

        by_id = {item.paper_id: item for item in evidence}
        for paper in papers:
            record = by_id.get(paper.id)
            if (
                record is None
                or canonical_evidence_key(record.title)
                != canonical_evidence_key(paper.title)
            ):
                continue
            self._verification_evidence_cache[
                (paper.id, EvidenceStore.content_hash(paper))
            ] = record

    def _record_retrieval_metrics(
        self,
        retrieval,
        *,
        count_routes: bool = True,
    ) -> None:
        if count_routes:
            self._metrics["verification_queries_executed"] += retrieval.requested_routes
        self._verification_paper_ids.update(
            paper.id
            for paper in retrieval.papers
        )

    def _record_prescreen_metrics(
        self,
        stats: tuple[int, int, int, int],
    ) -> None:
        considered, accepted, uncertain, rejected = stats
        self._metrics["verification_papers_prescreened"] += considered
        self._metrics["verification_papers_prescreen_accepted"] += accepted
        self._metrics["verification_papers_prescreen_uncertain"] += uncertain
        self._metrics["verification_papers_rejected_by_prescreen"] += rejected

    def _query_cache_key(
        self,
        query: SearchQuery,
    ) -> tuple[str, str, str, int, int]:
        provider = getattr(
            self.retriever.client,
            "provider_name",
            type(self.retriever.client).__name__,
        )
        return (
            canonical_evidence_key(query.text),
            str(provider).casefold(),
            RetrievalMode.BROAD_LEXICAL.value,
            self.per_query_limit,
            self.max_verification_papers,
        )

    @staticmethod
    def _candidate_cache_key(
        candidate: GapCandidate,
    ) -> tuple[object, ...]:
        basis = tuple(
            sorted(
                {
                    (
                        item.dimension,
                        canonical_evidence_key(item.value),
                    )
                    for item in candidate.landscape_basis
                }
            )
        )
        return (candidate.category, candidate.pattern_type, basis)

    def _retrieve_one_verification_query(
        self,
        query: VerificationQuery,
    ) -> MultiQueryResult:
        search_query = SearchQuery(
            text=query.query,
            strategy=query.strategy,
            source=query.source,
        )
        key = self._query_cache_key(search_query)
        cached = self._verification_query_cache.get(key)
        if cached is not None:
            self._metrics["verification_queries_cache_hits"] += 1
            self._record_retrieval_metrics(cached, count_routes=False)
            return cached

        result = self.retriever.retrieve_verification(
            [search_query],
            per_query_limit=self.per_query_limit,
            limit=self.max_verification_papers,
        )
        self._verification_query_cache[key] = result
        self._record_retrieval_metrics(result)
        return result

    def _retrieve_verification_queries(
        self,
        queries: Sequence[VerificationQuery],
        *,
        after_query: Callable[[Sequence[Paper]], bool] | None = None,
    ) -> MultiQueryResult:
        """Retrieve an ordered query plan with per-run reuse and early stopping."""

        self._metrics["verification_queries_planned"] += len(queries)
        papers: list[Paper] = []
        failures = []
        seen_ids: set[str] = set()
        no_new_streak = 0
        executed = 0

        for index, query in enumerate(queries):
            result = self._retrieve_one_verification_query(query)
            executed += 1
            failures.extend(result.failures)
            before = set(seen_ids)
            papers.extend(result.papers)
            unique = deduplicate_paper_models(papers)
            seen_ids.update(paper.id.casefold() for paper in unique)
            new_count = len(seen_ids - before)

            inspection_target = min(5, self.max_verification_papers)
            coverage_stop = (
                after_query(unique[:inspection_target])
                if after_query is not None
                else len(unique) >= inspection_target
            )
            if coverage_stop:
                self._metrics["verification_candidates_early_stopped"] += 1
                break

            if new_count == 0 and not result.failures:
                no_new_streak += 1
            else:
                no_new_streak = 0

            remaining = queries[index + 1:]
            if after_query is None and len(unique) >= inspection_target:
                self._metrics["verification_queries_skipped_redundant"] += len(remaining)
                break

            if no_new_streak >= 2 and remaining:
                if all(
                    any(
                        _query_overlap(next_query.query, prior.query) >= 0.8
                        for prior in queries[: index + 1]
                    )
                    for next_query in remaining
                ):
                    self._metrics["verification_queries_skipped_redundant"] += len(remaining)
                    break

        return MultiQueryResult(
            papers=deduplicate_paper_models(papers)[: self.max_verification_papers],
            failures=failures,
            requested_routes=executed,
        )

    def _extract_verification_evidence(
        self,
        papers: Sequence[Paper],
        records: dict[str, PaperEvidence],
        *,
        timing_key: str | None = None,
    ) -> list[PaperEvidence]:
        started = perf_counter() if timing_key else None
        requested = list(papers)[: self.max_verification_papers]
        pending: list[Paper] = []
        resolved: dict[str, PaperEvidence] = {}
        verifier_cache_hits = 0

        for paper in requested:
            key = (paper.id, EvidenceStore.content_hash(paper))
            cached = self._verification_evidence_cache.get(key)
            if cached is not None:
                resolved[paper.id] = cached
                verifier_cache_hits += 1
                continue

            caller_record = records.get(paper.id)
            if caller_record is not None and _record_can_be_reused_for_paper(
                caller_record,
                paper,
            ):
                self._verification_evidence_cache[key] = caller_record
                resolved[paper.id] = caller_record
                verifier_cache_hits += 1
                continue

            pending.append(paper)

        before = _extractor_metrics(self.extractor)
        extracted: list[PaperEvidence] = []
        if pending:
            get_many = getattr(
                self.extractor,
                "get_many_or_extract",
                None,
            )
            if get_many is None:
                get_many = self.extractor.extract_many
            # The authoritative extractor/store decides whether each request
            # is a memory hit, persistent hit, or new extraction.
            extracted = list(
                get_many(
                    pending,
                    limit=self.max_verification_papers,
                )
            )
        after = _extractor_metrics(self.extractor)

        for item in extracted:
            paper = next(
                (candidate for candidate in pending if candidate.id == item.paper_id),
                None,
            )
            if paper is None:
                continue
            self._verification_evidence_cache[
                (paper.id, EvidenceStore.content_hash(paper))
            ] = item
            resolved[item.paper_id] = item

        self._metrics["new_verification_extractions"] += max(
            0,
            after.get("new_evidence_extractions", 0)
            - before.get("new_evidence_extractions", 0),
        )
        if not before:
            self._metrics["new_verification_extractions"] += len(extracted)
        self._metrics["verification_papers_already_cached"] += verifier_cache_hits
        if before:
            self._metrics["verification_papers_already_cached"] += max(
                0,
                (
                    after.get("memory_cache_hits", 0)
                    + after.get("persistent_cache_hits", 0)
                )
                - (
                    before.get("memory_cache_hits", 0)
                    + before.get("persistent_cache_hits", 0)
                ),
            )
        else:
            # Caller-provided records are counted above only when their title
            # and provenance are compatible with the retrieved paper.
            pass
        result = [
            resolved[paper.id]
            for paper in requested
            if paper.id in resolved
        ]
        if timing_key and started is not None:
            self._timings[timing_key] += perf_counter() - started
        return result

    def verify_many(
        self,
        idea: ResearchIdea,
        candidates: Sequence[GapCandidate],
        evidence: Sequence[PaperEvidence],
    ) -> list[GapCandidate]:
        self.notices = []
        from src.analysis.gap_candidates import (
            candidate_anchor_score,
            prune_redundant_candidates,
        )

        valid_dimensions = {
            "problem",
            "population_or_setting",
            "method",
            "method_family",
            "dataset",
            "dataset_type",
            "baseline",
            "performance_metric",
            "efficiency_metric",
            "constraint",
            "limitation",
            "future_work",
            "outcome",
        }
        eligible: list[GapCandidate] = []
        for candidate in candidates:
            invalid_role = any(
                item.dimension not in valid_dimensions
                or item.dimension == "study_type"
                for item in candidate.landscape_basis
            )
            if invalid_role:
                self._metrics["candidate_rejected_invalid_role"] += 1
                continue

            anchor_score = candidate_anchor_score(candidate, idea)
            if len(candidate.landscape_basis) > 1 and anchor_score <= 0.0:
                self._metrics["candidate_rejected_unanchored"] += 1
                continue
            eligible.append(candidate)

        selected, pruned = prune_redundant_candidates(
            eligible,
            idea,
        )
        self._metrics["candidate_rejected_dominated"] += pruned
        self._metrics["candidate_hypotheses_after_pruning"] = len(selected)
        if len(selected) > self.max_verified_candidates:
            pruned += len(selected) - self.max_verified_candidates
            selected = selected[: self.max_verified_candidates]

        if pruned:
            self.notices.append(
                f"verification_pruning: skipped {pruned} redundant candidate hypotheses"
            )

        self._metrics["candidate_hypotheses_verified"] += len(selected)
        verified_selected = [
            self.verify(idea, candidate, evidence)
            for candidate in selected
        ]

        # The selected set is the final set. A dominated subset is not
        # equivalent to its stronger superset and must never be resurrected.
        return verified_selected

    def assess_idea(
        self,
        idea: ResearchIdea,
        landscape: LiteratureLandscape | None = None,
        evidence: Sequence[PaperEvidence] = (),
    ) -> IdeaAssessment:
        """Assess the complete user idea independently of candidate gaps."""

        queries = build_idea_verification_queries(
            idea,
            max_queries=self.max_queries,
        )

        # Initial evidence is already analyzed and therefore has stronger
        # provenance than a title-only external hit. Reuse it before building
        # or executing any direct verification request.
        initial_direct: list[PaperEvidence] = []
        initial_matched: dict[str, list[str]] = {}
        for record in evidence:
            complete, matched = _idea_match_strength(idea, record)
            if not complete or not _record_has_abstract_evidence(record):
                continue
            initial_direct.append(record)
            initial_matched[record.paper_id] = list(matched)

        if initial_direct:
            self._metrics["verification_papers_already_cached"] += len(
                initial_direct
            )
            supporting_evidence: list[GapEvidence] = []
            for record in initial_direct:
                for field_name, item in _idea_evidence(
                    idea,
                    record,
                    initial_matched[record.paper_id],
                )[:10]:
                    supporting_evidence.append(
                        GapEvidence(
                            paper_id=record.paper_id,
                            evidence_type=field_name,
                            value=item.value,
                            evidence_text=item.evidence_text,
                            study_type=record.study_type,
                            role="confirmed_direct_match",
                        )
                    )
            return IdeaAssessment(
                label="well_studied",
                rationale=(
                    "Existing analyzed evidence already supports the complete "
                    "research idea; external direct verification was not required."
                ),
                supporting_paper_ids=[item.paper_id for item in initial_direct],
                supporting_evidence=_unique_evidence(supporting_evidence),
                matched_facets=initial_matched,
                verification_queries=[],
                searched_paper_ids=[item.paper_id for item in initial_direct],
                coverage_notes=[
                    "Initial analyzed evidence was checked before external retrieval.",
                    "External direct verification was not required.",
                ],
            )

        if not queries:
            return IdeaAssessment(
                label="uncertain",
                rationale=(
                    "The complete idea did not yield a sufficiently "
                    "specific verification query."
                ),
                coverage_notes=[
                    "Direct idea verification did not execute."
                ],
            )

        started = perf_counter()
        retrieval = self._retrieve_verification_queries(queries)
        self._timings["direct_verification_retrieval"] += perf_counter() - started

        papers = _rank_for_idea(
            idea,
            retrieval.papers,
            self.lexical_scorer,
        )

        self._metrics["verification_evidence_requested"] += len(papers)
        screened, screen_stats = _pre_screen_papers_with_stats(
            papers,
            _idea_facet_groups(idea),
            self.lexical_scorer,
            _idea_all_terms(idea),
        )
        self._record_prescreen_metrics(screen_stats)

        # Targeted verification evidence takes precedence when the same paper
        # also appeared in the initial landscape.
        records = {
            item.paper_id: item
            for item in evidence
        }

        extracted = self._extract_verification_evidence(
            screened,
            records,
            timing_key="direct_verification_evidence_extraction",
        )
        records.update(
            {
                item.paper_id: item
                for item in extracted
            }
        )

        failures = [
            VerificationFailure(
                query=item.query,
                provider=item.provider,
                error=item.error,
            )
            for item in retrieval.failures
        ]

        extraction_failures = list(
            getattr(
                self.extractor,
                "failures",
                [],
            )
        )

        failures.extend(
            VerificationFailure(
                query="evidence_extraction",
                provider="evidence_extractor",
                error=str(item),
            )
            for item in extraction_failures
        )

        direct_ids: list[str] = []
        partial_ids: list[str] = []
        potential_ids: list[str] = []

        direct_evidence: list[GapEvidence] = []
        partial_evidence: list[GapEvidence] = []
        potential_evidence: list[GapEvidence] = []

        matched_facets: dict[str, list[str]] = {}

        retrieved_abstract_ids = {
            paper.id
            for paper in papers
            if paper.abstract
        }

        for record in records.values():
            complete, matched = _idea_match_strength(
                idea,
                record,
            )

            abstract_supported = (
                record.paper_id
                in retrieved_abstract_ids
                or _record_has_abstract_evidence(record)
            )

            if complete and abstract_supported:
                role: EvidenceRole = "confirmed_direct_match"
                target_ids = direct_ids
                target_evidence = direct_evidence

            elif matched and abstract_supported:
                role = "contextual_or_partial_support"
                target_ids = partial_ids
                target_evidence = partial_evidence

            elif matched or _title_may_match_idea(
                idea,
                record,
            ):
                role = "potential_match"
                target_ids = potential_ids
                target_evidence = potential_evidence

            else:
                continue

            target_ids.append(
                record.paper_id
            )

            matched_facets[
                record.paper_id
            ] = list(matched)

            evidence_items = _idea_evidence(
                idea,
                record,
                matched,
            )[:10]

            for field_name, item in evidence_items:
                target_evidence.append(
                    GapEvidence(
                        paper_id=record.paper_id,
                        evidence_type=field_name,
                        value=item.value,
                        evidence_text=item.evidence_text,
                        study_type=record.study_type,
                        role=role,
                    )
                )

            if (
                role == "potential_match"
                and not evidence_items
            ):
                target_evidence.append(
                    GapEvidence(
                        paper_id=record.paper_id,
                        evidence_type="title",
                        value=record.title,
                        evidence_text=record.title,
                        study_type=record.study_type,
                        role=role,
                    )
                )

        coverage_evidence = list(records.values())
        coverage_ok, coverage_detail = _idea_coverage(
            idea,
            coverage_evidence,
        )

        coverage_notes = [
            (
                f"Executed {retrieval.requested_routes} direct idea verification "
                f"queries and retrieved {len(papers)} unique papers."
            ),
            (
                "Direct verification used structured title/abstract "
                "evidence only."
            ),
            *coverage_detail,
        ]

        if extraction_failures:
            coverage_notes.append(
                f"Evidence extraction failed for "
                f"{len(extraction_failures)} verification papers."
            )

        if retrieval.failures:
            coverage_notes.append(
                "Verification provider failure prevents a positive "
                "gap assessment."
            )

        if any(
            not paper.abstract
            for paper in papers
        ):
            coverage_notes.append(
                "Some verification papers lacked abstracts; title-only "
                "matches were not treated as direct."
            )

        if direct_ids:
            label: GapAssessmentLabel = "well_studied"
            rationale = (
                "Targeted verification found a paper satisfying all "
                "explicitly represented research-idea requirements "
                "within the same study."
            )

        elif (
            failures
            or not coverage_ok
            or not papers
            or partial_ids
            or potential_ids
        ):
            label = "uncertain"
            rationale = _facet_aware_idea_rationale(
                idea,
                matched_facets,
                partial_ids,
                potential_ids,
                direct_ids,
                failures,
            )

        elif _positive_idea_signal(
            idea,
            landscape,
            coverage_evidence,
        ):
            label = "promising_gap"
            rationale = (
                "The analyzed evidence covers the explicit idea facets, "
                "targeted verification completed successfully, and no "
                "same-paper complete match was found."
            )

        else:
            label = "uncertain"
            rationale = _facet_aware_idea_rationale(
                idea,
                matched_facets,
                partial_ids,
                potential_ids,
                direct_ids,
                failures,
            )

        return IdeaAssessment(
            label=label,
            rationale=rationale,
            supporting_paper_ids=list(
                dict.fromkeys(
                    [
                        *direct_ids,
                        *partial_ids,
                        *potential_ids,
                    ]
                )
            ),
            supporting_evidence=_unique_evidence(
                [
                    *direct_evidence,
                    *partial_evidence,
                    *potential_evidence,
                ]
            ),
            counterexample_paper_ids=list(
                dict.fromkeys(direct_ids)
            ),
            counterexample_evidence=_unique_evidence(
                direct_evidence
            ),
            partial_match_paper_ids=list(
                dict.fromkeys(partial_ids)
            ),
            potential_match_paper_ids=list(
                dict.fromkeys(potential_ids)
            ),
            matched_facets=matched_facets,
            verification_queries=queries,
            searched_paper_ids=[
                paper.id
                for paper in papers
            ],
            coverage_notes=coverage_notes,
            failures=failures,
        )

    def verify(
        self,
        idea: ResearchIdea,
        candidate: GapCandidate,
        evidence: Sequence[PaperEvidence],
    ) -> GapCandidate:
        cache_key = self._candidate_cache_key(candidate)

        observed_paper_ids = _candidate_observed_papers(
            candidate,
            evidence,
        )
        if observed_paper_ids:
            self._metrics["observed_candidates_rejected_preverification"] += 1
            self._metrics["candidate_rejected_observed"] += 1
            verification = GapVerification(
                candidate_id=candidate.id,
                supporting_paper_ids=list(candidate.supporting_paper_ids),
                contradicting_paper_ids=observed_paper_ids,
                coverage_notes=[
                    "Candidate was already observed within one analyzed "
                    "paper; targeted verification was not required."
                ],
                label="well_studied",
                reason=(
                    "The complete candidate component set was already present "
                    "in one analyzed paper, so it was rejected as a missing "
                    "combination before external verification."
                ),
            )
            self._verification_outcome_cache[cache_key] = verification.model_copy(
                deep=True
            )
            return _apply_verification(candidate, verification)

        cached_outcome = self._verification_outcome_cache.get(cache_key)
        if cached_outcome is not None:
            self._metrics["verification_candidates_reused"] += 1
            self._metrics["candidate_verification_results_reused_exact"] += 1
            verification = cached_outcome.model_copy(deep=True)
            verification.candidate_id = candidate.id
            return _apply_verification(candidate, verification)

        queries = [
            query
            for query in candidate.verification_queries
            if query.candidate_id == candidate.id
            and _query_preserves_candidate_facets(
                query.query,
                candidate,
            )
        ][: self.max_queries]

        if not queries:
            queries = build_verification_queries(
                idea,
                candidate,
                max_queries=self.max_queries,
            )

        if not queries:
            return _apply_verification(
                candidate,
                GapVerification(
                    candidate_id=candidate.id,
                    verification_queries=[],
                    label="uncertain",
                    reason=(
                        "No candidate-specific counterexample query "
                        "could be constructed."
                    ),
                    coverage_notes=[
                        "Targeted verification did not execute."
                    ],
                ),
            )

        records = {
            item.paper_id: item
            for item in evidence
        }
        processed_ids: set[str] = set()
        extraction_failures: list[object] = []

        def inspect_after_query(current_papers: Sequence[Paper]) -> bool:
            new_papers = [
                paper
                for paper in current_papers
                if paper.id.casefold() not in processed_ids
            ]
            self._metrics["verification_evidence_requested"] += len(new_papers)
            screened, screen_stats = _pre_screen_papers_with_stats(
                new_papers,
                _candidate_facet_groups(candidate),
                self.lexical_scorer,
                _candidate_text(candidate, idea),
            )
            self._record_prescreen_metrics(screen_stats)

            extracted = self._extract_verification_evidence(
                screened,
                records,
                timing_key="candidate_verification_evidence_extraction",
            )
            records.update(
                {
                    item.paper_id: item
                    for item in extracted
                }
            )
            extraction_failures.extend(
                getattr(self.extractor, "failures", [])
            )
            processed_ids.update(
                paper.id.casefold()
                for paper in new_papers
            )

            direct = any(
                paper.abstract
                and (record := records.get(paper.id)) is not None
                and _direct_counterexample(candidate, record, idea)
                for paper in new_papers
            )
            useful = sum(
                records.get(paper.id) is not None
                for paper in new_papers
            )
            return direct or useful >= min(5, self.max_verification_papers)

        started = perf_counter()
        retrieval = self._retrieve_verification_queries(
            queries,
            after_query=inspect_after_query,
        )
        self._timings["candidate_verification_retrieval"] += perf_counter() - started
        executed_queries = queries[: retrieval.requested_routes]
        candidate.verification_queries = executed_queries

        papers = _rank_for_candidate(
            candidate,
            retrieval.papers,
            self.lexical_scorer,
        )

        failures = [
            VerificationFailure(
                query=failure.query,
                provider=failure.provider,
                error=failure.error,
            )
            for failure in retrieval.failures
        ]

        failures.extend(
            VerificationFailure(
                query="evidence_extraction",
                provider="evidence_extractor",
                error=str(item),
            )
            for item in extraction_failures
        )

        coverage_notes = [
            (
                f"Executed {len(executed_queries)} candidate-specific verification "
                f"queries and retrieved {len(papers)} unique papers."
            ),
            "Verification used structured title/abstract evidence only.",
        ]

        if extraction_failures:
            coverage_notes.append(
                f"Evidence extraction failed for "
                f"{len(extraction_failures)} verification papers."
            )

        if retrieval.failures:
            coverage_notes.append(
                "At least one verification provider route failed; "
                "failure is not treated as absence of counterexamples."
            )

        if any(
            not paper.abstract
            for paper in papers
        ):
            coverage_notes.append(
                "Some verification papers had no abstract; title-only "
                "matches remain potential, not confirmed."
            )

        potential: list[str] = []
        confirmed: list[str] = []
        inspected: list[GapEvidence] = []

        for paper in papers:
            record = records.get(
                paper.id
            )

            role: EvidenceRole | None = None
            items: list[
                tuple[str, EvidenceItem]
            ] = []

            if record is None:
                if _title_possible_match(
                    candidate,
                    paper,
                ):
                    role = "potential_contradiction"

                    items = [
                        (
                            "title",
                            EvidenceItem(
                                value=paper.title,
                                evidence_text=paper.title,
                                source="title",
                                confidence=0.4,
                            ),
                        )
                    ]

            else:
                direct = (
                    _record_has_abstract_evidence(record)
                    and _direct_counterexample(
                        candidate,
                        record,
                        idea,
                    )
                )

                contextual = _contextual_match(
                    candidate,
                    record,
                    idea,
                )

                if direct:
                    role = "confirmed_contradiction"
                elif contextual:
                    role = "potential_contradiction"
                elif (
                    not paper.abstract
                    and _title_possible_match(
                        candidate,
                        paper,
                    )
                ):
                    role = "potential_contradiction"

                items = _relevant_items(
                    candidate,
                    record,
                    idea,
                )

            if role is None:
                continue

            if role == "confirmed_contradiction":
                confirmed.append(
                    paper.id
                )
            else:
                potential.append(
                    paper.id
                )

            for evidence_type, item in items[:4]:
                inspected.append(
                    GapEvidence(
                        paper_id=paper.id,
                        evidence_type=evidence_type,
                        value=item.value,
                        evidence_text=item.evidence_text,
                        study_type=(
                            record.study_type
                            if record
                            else None
                        ),
                        role=role,
                    )
                )

        confirmed = list(
            dict.fromkeys(confirmed)
        )
        potential = list(
            dict.fromkeys(potential)
        )

        label, reason = _assessment(
            candidate,
            papers,
            records,
            confirmed,
            potential,
            failures,
        )

        verification = GapVerification(
            candidate_id=candidate.id,
            verification_queries=list(queries),
            searched_paper_ids=[
                paper.id
                for paper in papers
            ],
            supporting_paper_ids=list(
                candidate.supporting_paper_ids
            ),
            potential_contradiction_paper_ids=potential,
            contradicting_paper_ids=confirmed,
            evidence=_unique_evidence(inspected),
            coverage_notes=coverage_notes,
            failures=failures,
            label=label,
            reason=reason,
        )

        self._verification_outcome_cache[cache_key] = verification.model_copy(
            deep=True
        )
        return _apply_verification(
            candidate,
            verification,
        )

def _rank_for_idea(
    idea: ResearchIdea,
    papers: Sequence[Paper],
    scorer: LexicalScorer,
) -> list[Paper]:
    query = _idea_all_terms(
        idea
    )

    scores = scorer.score_many(
        query,
        papers,
    )

    return [
        paper
        for _, paper in sorted(
            zip(scores, papers),
            key=lambda pair: (
                -pair[0],
                pair[1].title.casefold(),
                pair[1].id.casefold(),
            ),
        )
    ]


def _rank_for_candidate(
    candidate: GapCandidate,
    papers: Sequence[Paper],
    scorer: LexicalScorer,
) -> list[Paper]:
    query = " ".join(
        [
            candidate.title,
            candidate.description,
            *(
                item.value
                for item in candidate.landscape_basis
            ),
        ]
    )

    scores = scorer.score_many(
        query,
        papers,
    )

    return [
        paper
        for _, paper in sorted(
            zip(scores, papers),
            key=lambda pair: (
                -pair[0],
                pair[1].title.casefold(),
                pair[1].id.casefold(),
            ),
        )
    ]


def _extractor_metrics(
    extractor,
) -> dict[str, int]:
    snapshot = getattr(extractor, "metrics_snapshot", None)
    if snapshot is None:
        return {}
    try:
        value = snapshot()
    except Exception:
        return {}
    return dict(value) if isinstance(value, dict) else {}

def _pre_screen_papers(
    papers: Sequence[Paper],
    phrases: Sequence[str] | Sequence[Sequence[str]],
    scorer: LexicalScorer,
    query: str,
) -> list[Paper]:
    """Return papers that are accepted or remain structurally ambiguous."""

    screened, _ = _pre_screen_papers_with_stats(
        papers,
        phrases,
        scorer,
        query,
    )
    return screened


def _pre_screen_papers_with_stats(
    papers: Sequence[Paper],
    phrases: Sequence[str] | Sequence[Sequence[str]],
    scorer: LexicalScorer,
    query: str,
) -> tuple[list[Paper], tuple[int, int, int, int]]:
    """Conservatively classify papers before expensive structured extraction.

    A paper is rejected only when it has no exact support for any requested
    facet group and its lexical score is also weak. Partial facet support and
    strong lexical similarity remain uncertain and are retained.
    """

    groups: list[tuple[set[str], ...]] = []
    for phrase in phrases:
        values = (phrase,) if isinstance(phrase, str) else tuple(phrase)
        alternatives = tuple(
            terms
            for value in values
            if value and (terms := _concept_tokens(value))
        )
        if alternatives:
            groups.append(alternatives)

    if len(papers) < 2 or not groups:
        # The low-volume path is not worth gating. Mark it uncertain so the
        # metrics describe that no aggressive decision was made.
        return list(papers), (len(papers), 0, len(papers), 0)

    scores = scorer.score_many(query, papers)
    retained: list[Paper] = []
    accepted = 0
    uncertain = 0
    rejected = 0

    for paper, score in zip(papers, scores):
        text_terms = _concept_tokens(
            f"{paper.title} {paper.abstract or ''}"
        )
        covered_groups = sum(
            any(required <= text_terms for required in alternatives)
            for alternatives in groups
        )

        if covered_groups == len(groups):
            accepted += 1
            retained.append(paper)
        elif covered_groups > 0 or score >= 0.75:
            # Partial support is intentionally uncertain: the extractor may
            # find cross-field evidence that a surface gate cannot see.
            uncertain += 1
            retained.append(paper)
        else:
            rejected += 1

    return retained, (len(papers), accepted, uncertain, rejected)

def _record_items(
    record: PaperEvidence,
) -> list[tuple[str, EvidenceItem]]:
    fields = (
        "research_objective",
        "population_or_setting",
        "method_or_intervention",
        "comparison_or_baseline",
        "data_or_modality",
        "datasets",
        "evaluation_metrics",
        "main_findings",
        "constraints",
        "limitations",
        "future_work",
    )

    result: list[
        tuple[str, EvidenceItem]
    ] = []

    for field_name in fields:
        value = getattr(
            record,
            field_name,
        )

        items = (
            value
            if isinstance(value, list)
            else [value]
            if value
            else []
        )

        result.extend(
            (
                field_name,
                item,
            )
            for item in items
        )

    return result


def _record_can_be_reused_for_paper(
    record: PaperEvidence,
    paper: Paper,
) -> bool:
    """Accept caller evidence only when its title and provenance still fit."""

    if canonical_evidence_key(record.title) != canonical_evidence_key(paper.title):
        return False

    items = _record_items(record)
    if not items:
        return False

    title = canonical_evidence_key(paper.title)
    abstract = canonical_evidence_key(paper.abstract or "")
    for _, item in items:
        source = title if item.source == "title" else abstract
        if canonical_evidence_key(item.evidence_text) not in source:
            return False
    return True


def _experimental_items(
    record: PaperEvidence,
) -> list[tuple[str, EvidenceItem]]:
    allowed = {
        "research_objective",
        "population_or_setting",
        "method_or_intervention",
        "comparison_or_baseline",
        "data_or_modality",
        "datasets",
        "evaluation_metrics",
        "main_findings",
        "constraints",
    }

    return [
        (field_name, item)
        for field_name, item in _record_items(record)
        if field_name in allowed
    ]


def _record_text(
    record: PaperEvidence,
) -> str:
    return " ".join(
        [
            record.title,
            *(
                f"{item.value} {item.evidence_text}"
                for _, item in _record_items(record)
            ),
        ]
    )


def _experimental_record_text(
    record: PaperEvidence,
) -> str:
    return " ".join(
        f"{item.value} {item.evidence_text}"
        for _, item in _experimental_items(record)
    )


def _record_has_abstract_evidence(
    record: PaperEvidence,
) -> bool:
    return any(
        item.source == "abstract"
        for _, item in _record_items(record)
    )

    "method": ("method_or_intervention",),

    },
    "data_or_modality": {

    },
    "comparison": {

    "outcome": {

}


def _facet_item_matches(
    value: str,
    item: EvidenceItem,
    dimension: str,
) -> bool:
    if dimension == "problem":
        return _problem_item_matches(value, item)
    if dimension == "method":
        return _method_item_matches(value, item)
    if dimension == "context":
        return _context_item_matches(value, item)
    if dimension == "data_or_modality":
        return _phrase_matches(value, f"{item.value} {item.evidence_text}")
    if dimension == "constraint":
        return _constraint_item_matches(value, item)
    if dimension == "comparison":
        return _comparison_item_matches(value, item)
    if dimension == "outcome":
        return _outcome_item_matches(value, item)
    return _phrase_matches(value, f"{item.value} {item.evidence_text}")


def _explicit_evidence_support(
    value: str,
    item: EvidenceItem,
) -> bool:
    """Recognize a requested phrase in another structured claim's text.

    This is deliberately containment of an already decomposed requirement,
    not an open-ended semantic similarity rule or a vocabulary lookup.
    """

    required = _concept_tokens(value)
    observed = _concept_tokens(
        f"{item.value} {item.evidence_text}"
    )
    return bool(required and required <= observed)


def _matching_facet_evidence(
    idea: ResearchIdea,
    values: Sequence[str],
    record: PaperEvidence,
    dimension: str,
) -> list[tuple[str, EvidenceItem]]:
    """Match each requested facet, preferring its source field.

    A structured field remains the strongest signal. Other fields can support
    the facet only when the claim text explicitly contains the decomposed
    requirement, allowing one structured sentence to support several facets.
    """

    groups = _idea_requirement_groups(idea, values, dimension)
    all_items = _record_items(record)
    primary_fields = _FACET_PRIMARY_FIELDS.get(dimension, ())
    primary_items = [
        (field_name, item)
        for field_name, item in all_items
        if field_name in primary_fields
    ]

    result: list[tuple[str, EvidenceItem]] = []
    used: set[tuple[str, str]] = set()

    for group in groups:
        match: tuple[str, EvidenceItem] | None = None

        for field_name, item in primary_items:
            if any(
                _requirement_item_matches(idea, value, item, dimension)
                for value in group
            ):
                match = (field_name, item)
                break

        if match is None:
            for field_name, item in all_items:
                if (
                    field_name not in _FACET_CROSS_FIELD_SUPPORT_FIELDS.get(
                        dimension,
                        set(),
                    )
                    or (field_name, item) in primary_items
                ):
                    continue
                if any(
                    _requirement_item_matches(idea, value, item, dimension)
                    or _explicit_evidence_support(value, item)
                    for value in group
                ):
                    match = (field_name, item)
                    break

        if match is None:
            continue

        key = (
            match[0],
            canonical_evidence_key(
                match[1].canonical_value or match[1].value
            ),
        )
        if key not in used:
            used.add(key)
            result.append(match)

    return result


def _facet_matches(
    idea: ResearchIdea,
    values: Sequence[str],
    record: PaperEvidence,
    dimension: str,
) -> bool:
    groups = _idea_requirement_groups(idea, values, dimension)
    if not groups:
        return False

    return all(
        _requirement_group_matches_record(
            idea,
            group,
            record,
            dimension,
        )
        for group in groups
    )


def _requirement_group_matches_record(
    idea: ResearchIdea,
    group: Sequence[str],
    record: PaperEvidence,
    dimension: str,
) -> bool:
    """Check one requirement group independently of display-evidence deduping."""

    all_items = _record_items(record)
    primary_fields = _FACET_PRIMARY_FIELDS.get(dimension, ())
    cross_fields = _FACET_CROSS_FIELD_SUPPORT_FIELDS.get(dimension, set())

    for field_name, item in all_items:
        if field_name in primary_fields and any(
            _requirement_item_matches(idea, value, item, dimension)
            for value in group
        ):
            return True

    for field_name, item in all_items:
        if field_name not in cross_fields:
            continue

        if any(
            _requirement_item_matches(idea, value, item, dimension)
            or _explicit_evidence_support(value, item)
            for value in group
        ):
            return True

    return False


def _record_supports_facet(
    value: str,
    record: PaperEvidence,
    dimension: str,
) -> bool:
    primary_fields = _FACET_PRIMARY_FIELDS.get(dimension, ())
    primary_items = [
        item
        for field_name, item in _record_items(record)
        if field_name in primary_fields
    ]
    if any(
        _facet_item_matches(value, item, dimension)
        for item in primary_items
    ):
        return True

    return any(
        _explicit_evidence_support(value, item)
        for _, item in _record_items(record)
    )

def _requirement_item_matches(
    idea: ResearchIdea,
    value: str,
    item: EvidenceItem,
    dimension: str,
) -> bool:
    """Apply canonical equality, then explicit/lexical matching fallbacks."""

    if _canonical_item_matches(idea, value, item, dimension):
        return True

    if _explicit_synonym_item_matches(idea, value, item, dimension):
        return True

    if _explicit_evidence_support(value, item):
        return True

    return _facet_item_matches(value, item, dimension)

def _idea_match_strength(
    idea: ResearchIdea,
    record: PaperEvidence,
) -> tuple[bool, list[str]]:
    groups: dict[str, bool] = {}

    problems = _idea_problems(
        idea
    )
    methods = _idea_methods(
        idea
    )
    context = _idea_context_values(
        idea
    )
    data_or_modality = [
        value
        for value in idea.data_or_modality
        if is_concrete_entity(value)
    ]
    constraints = _idea_constraints(
        idea
    )

    if problems:
        groups["problem"] = _problem_matches(
            problems,
            record,
            idea,
        )

    if methods:
        groups["method"] = _method_matches(
            methods,
            record,
            idea,
        )

    if context:
        groups["population_or_domain"] = _context_matches(
            context,
            record,
            idea,
        )

    if data_or_modality:
        groups["data_or_modality"] = _facet_matches(
            idea,
            data_or_modality,
            record,
            "data_or_modality",
        )

    if constraints:
        groups["constraint"] = _constraint_matches(
            idea,
            record,
        )

    if idea.comparison:
        groups["comparison"] = _comparison_matches(
            idea,
            record,
        )

    if idea.outcomes:
        groups["outcome"] = _outcome_matches(
            idea,
            record,
        )

    matched = [
        name
        for name, present in groups.items()
        if present
    ]

    return (
        bool(groups)
        and len(matched) == len(groups),
        matched,
    )

def _matching_requirement_items(
    idea: ResearchIdea,
    values: Sequence[str],
    items: Sequence[EvidenceItem],
    dimension: str,
) -> list[EvidenceItem]:
    groups = _idea_requirement_groups(
        idea,
        values,
        dimension,
    )

    result: list[EvidenceItem] = []

    for group in groups:
        for item in items:
            matched = any(
                _requirement_item_matches(idea, value, item, dimension)
                for value in group
            )

            if matched:
                result.append(
                    item
                )
                break

    return result


def _idea_evidence(
    idea: ResearchIdea,
    record: PaperEvidence,
    matched: Sequence[str],
) -> list[tuple[str, EvidenceItem]]:
    result: list[tuple[str, EvidenceItem]] = []
    requests = (
        ("problem", _idea_problems(idea), "problem"),
        ("method", _idea_methods(idea), "method"),
        ("population_or_domain", _idea_context_values(idea), "context"),
        ("data_or_modality", idea.data_or_modality, "data_or_modality"),
        ("constraint", _idea_constraints(idea), "constraint"),
        ("comparison", idea.comparison, "comparison"),
        ("outcome", idea.outcomes, "outcome"),
    )

    for facet, values, dimension in requests:
        if facet not in matched or not values:
            continue
        result.extend(
            _matching_facet_evidence(
                idea,
                values,
                record,
                dimension,
            )
        )

    unique: list[tuple[str, EvidenceItem]] = []
    seen: set[tuple[str, str]] = set()
    for field_name, item in result:
        key = (
            field_name,
            canonical_evidence_key(item.canonical_value or item.value),
        )
        if key in seen:
            continue
        seen.add(key)
        unique.append((field_name, item))
    return unique


# ---------------------------------------------------------------------------
# Coverage
# ---------------------------------------------------------------------------


def _idea_coverage(
    idea: ResearchIdea,
    evidence: Sequence[PaperEvidence],
) -> tuple[bool, list[str]]:
    notes = [
        (
            f"Analyzed {len(evidence)} evidence-bearing papers "
            "before direct idea assessment."
        )
    ]

    if len(evidence) < 2:
        notes.append(
            "Fewer than two evidence-bearing papers were available "
            "for coverage assessment."
        )

        return False, notes

    missing: list[str] = []

    checks = (
        (
            "problem",
            _idea_problems(idea),
            "problem",
            lambda value, record: _problem_matches(
                [value],
                record,
                idea,
            ),
        ),
        (
            "method",
            _idea_methods(idea),
            "method",
            lambda value, record: _method_matches(
                [value],
                record,
                idea,
            ),
        ),
        (
            "data_or_modality",
            idea.data_or_modality,
            "data_or_modality",
            lambda value, record: _facet_matches(
                idea,
                [value],
                record,
                "data_or_modality",
            ),
        ),
        (
            "comparison",
            idea.comparison,
            "comparison",
            lambda value, record: _comparison_matches(
                idea.model_copy(
                    update={
                        "comparison": [value]
                    }
                ),
                record,
            ),
        ),
        (
            "outcome",
            idea.outcomes,
            "outcome",
            lambda value, record: _outcome_matches(
                idea.model_copy(
                    update={
                        "outcomes": [value]
                    }
                ),
                record,
            ),
        ),
    )

    for name, values, dimension, matcher in checks:
        groups = _idea_requirement_groups(
            idea,
            values,
            dimension,
        )

        if groups and not all(
            any(
                matcher(
                    value,
                    record,
                )
                for value in group
                for record in evidence
            )
            for group in groups
        ):
            missing.append(
                name
            )

    context = _idea_context_values(
        idea
    )

    context_groups = _idea_requirement_groups(
        idea,
        context,
        "context",
    )

    if context_groups and not all(
        any(
            _context_matches(
                [value],
                record,
                idea,
            )
            for value in group
            for record in evidence
        )
        for group in context_groups
    ):
        missing.append(
            "population_or_domain"
        )

    constraint_groups = _idea_requirement_groups(
        idea,
        _idea_constraints(idea),
        "constraint",
    )

    if constraint_groups and not all(
        any(
            _facet_matches(
                idea,
                [value],
                record,
                "constraint",
            )
            for value in group
            for record in evidence
        )
        for group in constraint_groups
    ):
        missing.append(
            "constraint"
        )

    if missing:
        notes.append(
            "Important idea facets without explicit structured coverage: "
            + ", ".join(missing)
            + "."
        )

    return (
        not missing,
        notes,
    )


def _positive_idea_signal(
    idea: ResearchIdea,
    landscape: LiteratureLandscape | None,
    evidence: Sequence[PaperEvidence],
) -> bool:
    if (
        landscape is None
        or landscape.total_papers < 2
        or len(evidence) < 2
    ):
        return False

    coverage_ok, _ = _idea_coverage(
        idea,
        evidence,
    )

    return coverage_ok


def _requested_facet_names(
    idea: ResearchIdea,
) -> list[str]:
    facets: list[tuple[str, Sequence[str]]] = [
        ("problem", _idea_problems(idea)),
        ("method", _idea_methods(idea)),
        ("population_or_domain", _idea_context_values(idea)),
        ("data_or_modality", idea.data_or_modality),
        ("constraint", _idea_constraints(idea)),
        ("comparison", idea.comparison),
        ("outcome", idea.outcomes),
    ]
    return [
        name
        for name, values in facets
        if _idea_requirement_groups(idea, values, name if name != "population_or_domain" else "context")
    ]


def _facet_aware_idea_rationale(
    idea: ResearchIdea,
    matched_facets: dict[str, list[str]],
    partial_ids: Sequence[str],
    potential_ids: Sequence[str],
    direct_ids: Sequence[str],
    failures: Sequence[VerificationFailure],
) -> str:
    requested = _requested_facet_names(idea)
    covered = [
        name
        for name in requested
        if any(name in facets for facets in matched_facets.values())
    ]
    uncovered = [name for name in requested if name not in covered]

    if not requested:
        return (
            "Structured facet information was unavailable, so the complete "
            "research idea could not be assessed conservatively."
        )

    parts = [
        "Structured verification covered: "
        + (", ".join(covered) if covered else "none")
        + ".",
    ]

    if uncovered:
        parts.append(
            "Currently unconfirmed components: "
            + ", ".join(uncovered)
            + "."
        )
    else:
        parts.append(
            "All requested components appeared in the inspected evidence, "
            "but the complete combination was not confirmed within one paper."
        )

    if partial_ids:
        parts.append(
            f"Partial structured matches remained for {len(partial_ids)} paper(s)."
        )
    if potential_ids:
        parts.append(
            f"Potential title or weak matches remained for {len(potential_ids)} paper(s)."
        )
    if not direct_ids:
        parts.append("No complete same-paper match was confirmed.")
    if failures:
        parts.append(
            f"{len(failures)} retrieval or extraction failure(s) also limit coverage."
        )

    return " ".join(parts)

def _basis_requirement_matches(
    dimension: str,
    value: str,
    record: PaperEvidence,
) -> bool:
    if dimension == "problem":
        return _record_supports_facet(value, record, "problem")

    if dimension in {
        "method",
        "method_family",
    }:
        return _record_supports_facet(value, record, "method")

    if dimension in {
        "population",
        "population_or_setting",
        "setting",
    }:
        return _record_supports_facet(value, record, "context")

    if dimension == "constraint":
        return _record_supports_facet(value, record, "constraint")

    if dimension == "data_or_modality":
        return _record_supports_facet(value, record, "data_or_modality")

    if dimension == "dataset":
        target = normalize_dataset(
            value
        )

        return bool(
            target
            and any(
                normalize_dataset(
                    item.value
                ) == target
                for item in record.datasets
            )
        )

    if dimension == "dataset_type":
        target = canonical_evidence_key(
            value
        )

        return target in {
            canonical_evidence_key(
                item
            )
            for item in dataset_types(record)
        }

    if dimension in {
        "baseline",
        "comparison",
    }:
        return any(
            _comparison_item_matches(
                value,
                item,
            )
            for item in record.comparison_or_baseline
        )

    if dimension in {
        "performance_metric",
        "efficiency_metric",
        "metric",
    }:
        target = normalize_metric(
            value
        )

        return bool(
            target
            and any(
                normalize_metric(
                    item.value
                ) == target
                for item in record.evaluation_metrics
            )
        )

    if dimension == "outcome":
        return any(
            _outcome_item_matches(
                value,
                item,
            )
            for item in record.main_findings
        )

    if dimension == "study_type":
        return (
            canonical_evidence_key(
                record.study_type
            )
            == canonical_evidence_key(value)
        )

    if dimension == "limitation":
        return any(
            _phrase_matches(
                value,
                f"{item.value} {item.evidence_text}",
            )
            for item in record.limitations
        )

    if dimension == "future_work":
        return any(
            _phrase_matches(
                value,
                f"{item.value} {item.evidence_text}",
            )
            for item in record.future_work
        )

    return False


def _record_matches_idea_topic(
    idea: ResearchIdea,
    record: PaperEvidence,
) -> bool:
    """Keep candidate verification tied to the original research context."""

    problems = _idea_problems(
        idea
    )

    if (
        problems
        and record.research_objective
        and any(
            _problem_item_matches(
                value,
                record.research_objective,
            )
            for value in problems
        )
    ):
        return True

    context = _idea_context_values(
        idea
    )

    if context and any(
        _context_item_matches(
            value,
            item,
        )
        for value in context
        for item in record.population_or_setting
    ):
        return True

    methods = _idea_methods(
        idea
    )

    if methods and any(
        _method_item_matches(
            value,
            item,
        )
        for value in methods
        for item in record.method_or_intervention
    ):
        return True

    if idea.data_or_modality and any(
        _facet_item_matches(value, item, "data_or_modality")
        for value in idea.data_or_modality
        for item in [*record.data_or_modality, *record.datasets]
    ):
        return True

    if not (
        problems
        or context
        or methods
        or idea.data_or_modality
    ):
        return bool(
            set(_tokens(_idea_all_terms(idea)))
            & set(_tokens(_record_text(record)))
        )

    return False


def _strict_basis_requirement_matches(
    dimension: str,
    value: str,
    record: PaperEvidence,
) -> bool:
    """Match a candidate component only in its typed source role."""

    if dimension == "problem":
        return bool(
            record.research_objective
            and _problem_item_matches(value, record.research_objective)
        )

    if dimension in {"method", "method_family"}:
        return any(
            _method_item_matches(value, item)
            for item in record.method_or_intervention
        )

    if dimension in {"population", "population_or_setting", "setting"}:
        return any(
            _context_item_matches(value, item)
            for item in record.population_or_setting
        )

    if dimension == "constraint":
        return any(
            _constraint_item_matches(value, item)
            for item in record.constraints
        )

    if dimension == "data_or_modality":
        return any(
            _facet_item_matches(value, item, "data_or_modality")
            for item in [*record.data_or_modality, *record.datasets]
        )

    if dimension == "dataset":
        target = normalize_dataset(value)
        return bool(
            target
            and any(
                normalize_dataset(item.value) == target
                for item in record.datasets
            )
        )

    if dimension == "dataset_type":
        target = canonical_evidence_key(value)
        return target in {
            canonical_evidence_key(item)
            for item in dataset_types(record)
        }

    if dimension in {"baseline", "comparison"}:
        return any(
            _comparison_item_matches(value, item)
            for item in record.comparison_or_baseline
        )

    if dimension in {"metric", "performance_metric", "efficiency_metric"}:
        target = normalize_metric(value)
        return bool(
            target
            and any(
                normalize_metric(item.value) == target
                for item in record.evaluation_metrics
            )
        )

    if dimension == "outcome":
        return any(
            _outcome_item_matches(value, item)
            for item in record.main_findings
        )

    return False


def _candidate_observed_papers(
    candidate: GapCandidate,
    evidence: Sequence[PaperEvidence],
) -> list[str]:
    """Find same-paper observations for combination candidates only."""

    if candidate.pattern_type != "combination_gap":
        return []

    basis = [
        item
        for item in candidate.landscape_basis
        if is_concrete_entity(item.value)
    ]
    if len(basis) < 2:
        return []

    return [
        record.paper_id
        for record in evidence
        if all(
            _strict_basis_requirement_matches(
                item.dimension,
                item.value,
                record,
            )
            for item in basis
        )
    ]

# Candidate counterexample confirmation

def _explicit_comparison_covers_families(
    candidate_families: set[str],
    record: PaperEvidence,
) -> bool:
    """Require every candidate method family in one explicit comparison."""

    methods = [
        item
        for item in record.method_or_intervention
        if is_concrete_entity(item.value)
    ]

    baselines = [
        item
        for item in record.comparison_or_baseline
        if is_concrete_entity(item.value)
    ]

    method_families = {
        normalize_method_family(
            item.value
        )
        for item in methods
    }

    baseline_families = {
        normalize_method_family(
            item.value
        )
        for item in baselines
    }

    method_families.discard("")
    baseline_families.discard("")

    observed = (
        method_families
        | baseline_families
    )

    if not candidate_families <= observed:
        return False

    # Standard focal-method vs baseline representation.
    if (
        candidate_families
        & method_families
        and candidate_families
        & baseline_families
    ):
        return True

    # Neutral head-to-head studies can place all compared methods in the
    # primary-method field. Require one structured evidence statement that
    # explicitly expresses the comparison and names every family through one
    # of its actually extracted method entities.
    family_aliases: dict[
        str,
        list[str],
    ] = {}

    for item in [
        *methods,
        *baselines,
    ]:
        family = normalize_method_family(
            item.value
        )

        if family in candidate_families:
            family_aliases.setdefault(
                family,
                []
            ).append(
                item.value
            )

    relationship_items = [
        *methods,
        *baselines,
        *record.main_findings,
    ]

    for item in relationship_items:
        text = (
            f"{item.value} "
            f"{item.evidence_text}"
        )

        if not _COMPARISON_RELATION_PATTERN.search(
            text
        ):
            continue

        if all(
            any(
                _phrase_matches(
                    alias,
                    text,
                )
                for alias in family_aliases.get(
                    family,
                    []
                )
            )
            for family in candidate_families
        ):
            return True

    return False


def _narrow_dataset_counterexample(
    candidate: GapCandidate,
    record: PaperEvidence,
) -> bool:
    dominant_types = {
        canonical_evidence_key(
            item.value
        )
        for item in candidate.landscape_basis
        if item.dimension == "dataset_type"
        and is_concrete_entity(item.value)
    }

    if not dominant_types:
        return False

    observed_types = {
        canonical_evidence_key(
            value
        )
        for value in dataset_types(record)
        if is_concrete_entity(value)
    }

    return bool(
        observed_types
        - dominant_types
    )


def _resolution_counterexample(
    candidate: GapCandidate,
    record: PaperEvidence,
) -> bool:
    """Require actual study/evaluation evidence, not another limitation."""

    experimental = _experimental_items(
        record
    )

    text = " ".join(
        f"{item.value} {item.evidence_text}"
        for _, item in experimental
    )

    if not _STUDY_ACTION_PATTERN.search(
        text
    ):
        return False

    basis_terms = [
        item.value
        for item in candidate.landscape_basis
        if item.dimension in {
            "limitation",
            "future_work",
            "problem",
            "population_or_setting",
            "method",
            "method_family",
            "constraint",
            "dataset",
            "dataset_type",
            "outcome",
        }
    ]

    if not basis_terms:
        basis_terms = [
            candidate.title,
        ]

    return any(
        _phrase_matches(
            value,
            text,
        )
        for value in basis_terms
    )


def _conflict_resolution_counterexample(
    candidate: GapCandidate,
    record: PaperEvidence,
) -> bool:
    text = _experimental_record_text(
        record
    )

    if not _CONFLICT_RESOLUTION_PATTERN.search(
        text
    ):
        return False

    candidate_terms = set(
        _tokens(candidate.title)
    )

    record_terms = set(
        _tokens(text)
    )

    return bool(
        candidate_terms
        & record_terms
    )


def _direct_counterexample(
    candidate: GapCandidate,
    record: PaperEvidence,
    idea: ResearchIdea,
) -> bool:
    """Confirm only pattern-specific evidence with correct structured fields."""

    if not _record_matches_idea_topic(
        idea,
        record,
    ):
        return False

    pattern = candidate.pattern_type

    if pattern == "combination_gap":
        basis = [
            item
            for item in candidate.landscape_basis
            if is_concrete_entity(item.value)
        ]

        if len(basis) < 2:
            return False

        return all(
            _basis_requirement_matches(
                item.dimension,
                item.value,
                record,
            )
            for item in basis
        )

    if pattern == "missing_comparison":
        families = {
            normalize_method_family(
                item.value
            )
            for item in candidate.landscape_basis
            if item.dimension in {
                "method",
                "method_family",
            }
            and is_concrete_entity(item.value)
        }

        families.discard("")

        return (
            len(families) >= 2
            and _explicit_comparison_covers_families(
                families,
                record,
            )
        )

    if pattern == "narrow_dataset_setting":
        return _narrow_dataset_counterexample(
            candidate,
            record,
        )

    if pattern in {
        "repeated_limitation",
        "repeated_future_work",
        "underrepresented_population",
        "limited_external_validation",
        "limited_real_world_validation",
        "evaluation_gap",
        "replication_gap",
        "method_domain_transfer",
    }:
        return _resolution_counterexample(
            candidate,
            record,
        )

    if pattern == "comparable_conflict":
        return _conflict_resolution_counterexample(
            candidate,
            record,
        )

    # Unknown/unsupported gap patterns are never promoted to confirmed
    # contradictions merely by lexical similarity.
    return False

# Candidate contextual evidence
# ---------------------------------------------------------------------------


def _candidate_text(
    candidate: GapCandidate,
    idea: ResearchIdea,
) -> str:
    return " ".join(
        [
            candidate.title,
            candidate.description,
            *(
                item.value
                for item in candidate.landscape_basis
            ),
            *idea.problem,
            *idea.domain,
            *idea.population,
            *idea.intervention_or_method,
        ]
    )


def _contextual_match(
    candidate: GapCandidate,
    record: PaperEvidence,
    idea: ResearchIdea,
) -> bool:
    """Require meaningful candidate evidence, not generic vocabulary overlap."""

    if not _record_matches_idea_topic(
        idea,
        record,
    ):
        return False

    basis = [
        item
        for item in candidate.landscape_basis
        if is_concrete_entity(item.value)
    ]

    if basis:
        return any(
            _basis_requirement_matches(
                item.dimension,
                item.value,
                record,
            )
            for item in basis
        )

    generic_terms = {
        "method",
        "methods",
        "model",
        "models",
        "constraint",
        "constraints",
        "task",
        "study",
        "studies",
        "evaluation",
        "comparison",
        "dataset",
        "datasets",
        "population",
        "setting",
        "approach",
        "approaches",
    }

    candidate_terms = (
        set(
            _tokens(
                candidate.title
                + " "
                + candidate.description
            )
        )
        - generic_terms
    )

    record_terms = (
        set(
            _tokens(
                _record_text(record)
            )
        )
        - generic_terms
    )

    return bool(
        candidate_terms
        and record_terms
        and len(
            candidate_terms
            & record_terms
        )
        >= 2
    )


def _relevant_items(
    candidate: GapCandidate,
    record: PaperEvidence,
    idea: ResearchIdea,
) -> list[
    tuple[
        str,
        EvidenceItem,
    ]
]:
    terms = set(
        _tokens(
            _candidate_text(
                candidate,
                idea,
            )
        )
    )

    result: list[
        tuple[
            str,
            EvidenceItem,
        ]
    ] = []

    for field_name, item in _record_items(
        record
    ):
        item_terms = set(
            _tokens(
                f"{item.value} {item.evidence_text}"
            )
        )

        if terms & item_terms:
            result.append(
                (
                    field_name,
                    item,
                )
            )

    if result:
        return result

    # Keep some inspected structured evidence for transparency when semantic
    # normalization established the match without literal token overlap.
    for field_name, item in _experimental_items(
        record
    )[:2]:
        result.append(
            (
                field_name,
                item,
            )
        )

    return result

def _title_possible_match(
    candidate: GapCandidate,
    paper: Paper,
) -> bool:
    candidate_terms = set(
        _tokens(
            " ".join(
                [
                    candidate.title,
                    candidate.description,
                    *(
                        item.value
                        for item in candidate.landscape_basis
                    ),
                ]
            )
        )
    )

    title_terms = set(
        _tokens(paper.title)
    )

    required = max(
        2,
        min(
            4,
            len(candidate_terms),
        ),
    )

    return (
        len(
            candidate_terms
            & title_terms
        )
        >= required
    )


def _title_may_match_idea(
    idea: ResearchIdea,
    record: PaperEvidence,
) -> bool:
    idea_terms = set(
        _tokens(
            _idea_all_terms(idea)
        )
    )

    title_terms = set(
        _tokens(record.title)
    )

    return bool(
        idea_terms
        and len(
            idea_terms
            & title_terms
        )
        >= 2
    )


# ---------------------------------------------------------------------------
# Candidate assessment
# ---------------------------------------------------------------------------


def _assessment(
    candidate: GapCandidate,
    papers: Sequence[Paper],
    records: dict[str, PaperEvidence],
    confirmed: Sequence[str],
    potential: Sequence[str],
    failures: Sequence[VerificationFailure],
) -> tuple[
    GapAssessmentLabel,
    str,
]:
    if failures:
        return (
            "uncertain",
            (
                "Targeted verification had provider or extraction failures, "
                "so absence of a counterexample cannot be interpreted safely."
            ),
        )

    if papers and not records:
        return (
            "uncertain",
            (
                "Verification retrieved papers but no usable structured "
                "evidence was available for inspection."
            ),
        )

    if papers and all(
        not records.get(
            paper.id
        )
        or not _record_has_abstract_evidence(
            records[paper.id]
        )
        for paper in papers
    ):
        return (
            "uncertain",
            (
                "Available verification evidence is title-only or lacks "
                "usable abstract evidence."
            ),
        )

    if confirmed:
        return (
            "well_studied",
            (
                "Targeted verification found same-paper structured evidence "
                "that directly addresses the candidate hypothesis."
            ),
        )

    if potential:
        return (
            "uncertain",
            (
                "Targeted verification found possible matches, but the "
                "structured evidence was not sufficient for confirmation."
            ),
        )

    if not papers:
        return (
            "uncertain",
            (
                "No papers were retrieved by targeted verification; "
                "retrieval absence is not evidence of a research gap."
            ),
        )

    return (
        "promising_gap",
        (
            "Targeted counterexample searches completed without a confirmed "
            "direct counterexample. This remains a qualified evidence-based "
            "hypothesis, not a claim of global novelty."
        ),
    )

def _unique_evidence(
    items: Sequence[GapEvidence],
) -> list[GapEvidence]:
    result: list[GapEvidence] = []
    seen: set[
        tuple[
            str,
            str,
            str,
            str,
        ]
    ] = set()

    for item in items:
        key = (
            item.paper_id,
            item.evidence_type,
            canonical_evidence_key(
                item.value
            ),
            item.role,
        )

        if key in seen:
            continue

        seen.add(
            key
        )
        result.append(
            item
        )

    return result


def _apply_verification(
    candidate: GapCandidate,
    verification: GapVerification,
) -> GapCandidate:
    candidate.verification = verification
    candidate.final_label = verification.label

    candidate.potentially_contradicting_paper_ids = list(
        verification.potential_contradiction_paper_ids
    )

    candidate.contradicting_paper_ids = list(
        verification.contradicting_paper_ids
    )

    candidate.potentially_contradicting_evidence = [
        item
        for item in verification.evidence
        if item.role
        == "potential_contradiction"
    ]

    candidate.contradicting_evidence = [
        item
        for item in verification.evidence
        if item.role
        == "confirmed_contradiction"
    ]

    candidate.verification_status = "verified"

    return candidate
